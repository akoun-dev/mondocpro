   176→   176→// Formatage lisible : +2250701020304 → +225 07 01 02 03 04 ; 01020304 → 01 02 03 04
   177→   177→function formatPhoneDisplay(phone: string): string {
   178→   178→    const cleaned = phone.replace(/[\s.-]/g, "")
   179→   179→    const hasPlus = cleaned.startsWith("+")
   180→   180→    const digits = hasPlus ? cleaned.slice(1) : cleaned
   181→   181→    if (!/^\d{8,15}$/.test(digits)) return phone
   182→   182→
   183→   183→    if (hasPlus && digits.length > 10) {
   184→   184→        const countryCode = digits.slice(0, digits.length - 10)
   185→   185→        const local = digits.slice(-10)
   186→   186→        return `+${countryCode} ${local.replace(/(\d{2})(?=\d)/g, "$1 ")}`
   187→   187→    }
   188→   188→    return `${hasPlus ? "+" : ""}${digits.replace(/(\d{2})(?=\d)/g, "$1 ")}`
   189→   189→}
   190→   190→
   191→   191→function getInitials(fullName: string): string {
   192→   192→    const parts = fullName.trim().split(/\s+/).filter(Boolean)
   193→   193→    if (parts.length === 0) return "?"
   194→   194→    const first = parts[0].charAt(0)
   195→   195→    const second = parts.length > 1 ? parts[parts.length - 1].charAt(0) : ""
   196→   196→    return `${first}${second}`.toUpperCase()
   197→   197→}

   201→   201→}
