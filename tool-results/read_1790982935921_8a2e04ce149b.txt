   171→// Formatage lisible : +2250701020304 → +225 07 01 02 03 04 ; 01020304 → 01 02 03 04
   172→function formatPhoneDisplay(phone: string): string {
   173→    const cleaned = phone.replace(/[\s.-]/g, "")
   174→    const hasPlus = cleaned.startsWith("+")
   175→    const digits = hasPlus ? cleaned.slice(1) : cleaned
   176→    if (!/^\d{8,15}$/.test(digits)) return phone
   177→
   178→    if (hasPlus && digits.length > 10) {
   179→        const countryCode = digits.slice(0, digits.length - 10)
   180→        const local = digits.slice(-10)
   181→        return `+${countryCode} ${local.replace(/(\d{2})(?=\d)/g, "$1 ")}`
   182→    }
   183→    return `${hasPlus ? "+" : ""}${digits.replace(/(\d{2})(?=\d)/g, "$1 ")}`
   184→}
   185→
   186→function getInitials(fullName: string): string {
   187→    const parts = fullName.trim().split(/\s+/).filter(Boolean)
   188→    if (parts.length === 0) return "?"
   189→    const first = parts[0].charAt(0)
   190→    const second = parts.length > 1 ? parts[parts.length - 1].charAt(0) : ""
   191→    return `${first}${second}`.toUpperCase()
   192→}
   193→
   194→function getFirstName(fullName: string): string {
   195→    return fullName.trim().split(/\s+/)[0] ?? ""
   196→}
