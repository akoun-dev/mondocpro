   176→// Formatage lisible : +2250701020304 → +225 07 01 02 03 04 ; 01020304 → 01 02 03 04
   177→function formatPhoneDisplay(phone: string): string {
   178→    const cleaned = phone.replace(/[\s.-]/g, "")
   179→    const hasPlus = cleaned.startsWith("+")
   180→    const digits = hasPlus ? cleaned.slice(1) : cleaned
   181→    if (!/^\d{8,15}$/.test(digits)) return phone
   182→
   183→    if (hasPlus && digits.length > 10) {
   184→        const countryCode = digits.slice(0, digits.length - 10)
   185→        const local = digits.slice(-10)
   186→        return `+${countryCode} ${local.replace(/(\d{2})(?=\d)/g, "$1 ")}`
   187→    }
   188→    return `${hasPlus ? "+" : ""}${digits.replace(/(\d{2})(?=\d)/g, "$1 ")}`
   189→}
   190→
   191→function getInitials(fullName: string): string {
   192→    const parts = fullName.trim().split(/\s+/).filter(Boolean)
   193→    if (parts.length === 0) return "?"
   194→    const first = parts[0].charAt(0)
   195→    const second = parts.length > 1 ? parts[parts.length - 1].charAt(0) : ""
   196→    return `${first}${second}`.toUpperCase()
   197→}
   198→
   199→function getFirstName(fullName: string): string {
   200→    return fullName.trim().split(/\s+/)[0] ?? ""
   201→}
